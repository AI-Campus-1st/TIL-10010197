#중간값 찾기

def get_middle_char(word):
    if len(word) % 2:
        mid_idx = len(word)//2 
        ans = word[mid_idx]
    else: 
        mid_idx = len(word)//2
        ans = word[round(mid_idx)-1] + word[round(mid_idx)]
    return ans

get_middle_char('study') # => u
get_middle_char('python') # => th

# 가변 인자 리스트를 사용하여, 개수가 정해지지 않은 여러 정수들을 전달 받아 해당 정수들의 평균 값을 반환하는 my_avg 함수를 작성하시오
# my_avg(77, 83, 95, 80, 70) #=> 81.0


def my_avg(*args):
    sum = 0
    avg = 0
    for n in args:
        print(n)
        sum +=n

    avg = sum / len(args)
    return avg 
print(my_avg(77,83,95,80,70))
