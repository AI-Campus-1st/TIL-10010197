class Animal:
	def __init__(self, name):
		self.name = name
	
	def walk(self):
		print(f'{self.name}! 걷는다!')
	
	def eat(self):
		print(f'{self.name}! 먹는다!')

class Dog(Animal):

	def walk(self):
		print(f'{self.name}! 달린다!')
	
	def bark(self):
		print(f'{self.name}! 짖는다!')

class Bird(Animal):

    def fly(self):
        print(f'{self.name}! 푸드덕!')

dog = Dog('멍멍이')
dog.walk() 
dog.bark() 

bird = Bird('구구')
bird.walk() 
bird.eat() 
bird.fly()

# Payment 클래스
class Payment:
    
    def __init__(self,owner, balance):
        self.owner = owner
        self.balance = balance

    
    # 수수료 메서드 (fee)를 정의하여 결제금액에 대한 수수료를 반환하는 추상메서드를 만든다. (Payment 클래스에서는 구현하지 않음)
    
    def fee (self, amount):#amount와 연관되므로 추가
       pass
   
       
    # 총 차감액 = amount + self.fee(amount)
    # 잔액이 총 차감액보다 적으면 ValueError("잔액 부족")를 발생시킨다. (raise 활용)
    # 충분하면 잔액에서 총 차감액을 빼고, {"amount": amount, "fee": 수수료, "balance": 결제후잔액} 딕셔너리를 반환한다.

    def pay(self,amount):
        fee =self.fee(amount) #fee 등록
        total_sub = amount +fee
        if self.balance < total_sub:
            raise ValueError("잔액 부족")
        self.balance -= total_sub
        
        return {"amount": amount, "fee": fee, "balance": self.balance}
    

   class CardPayment(Payment):
     def fee(self, amount):#amount 연관되므로 등록
         return  int(amount * 0.025)
class MobilePayment(Payment):
     def fee(self,amount):
         return  100
class PointPayment(Payment):
     def fee(self,amount):
         return  0
     def pay(self,amount):
          result = super().pay(amount)
          result["point used"] = True
          return result

    # 실행 코드
print(CardPayment("홍길동", 20000).pay(10000))    # 수수료 250
print(MobilePayment("김철수", 20000).pay(10000))  # 수수료 100
print(PointPayment("이영희", 20000).pay(10000))   # 수수료 0, point_used True
