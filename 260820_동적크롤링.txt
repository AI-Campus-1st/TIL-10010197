import json 
import requests
URL = 'https://ac.search.naver.com/nx/ac'
PARAMS={
    
    'q_enc': 'UTF-8',
    'st': '100',
    #'_callback': '_jsonp_9'
        
    }

def get_related_keywords(keyword):
    res = requests.get(URL, params={**PARAMS, 'q':keyword})
    data= res.json()

    related_keywords = []
    for item in data['items'][0]: #대괄호 하나 지우기 
        related_keywords.append(item[0])
    return related_keywords

#변환 방법1. 함수
def get_dow_eng2kor(dow_eng):
    match dow_eng:
        case 'MONDAY':
            return '월'
        case 'TUESDAY':
            return '화'
        case 'WEDNSDAY':
            return '수'
        case 'THURSDAY':
            return '목'
        case 'FRIDAY':
            return '금'
        case 'SATURDAY':
            return '토'
        case 'SUNDAY':
            return '일'
    return None #요일을 한글로 변환함수
